import requests

def weather():
  city = input("enter city name:")
  url= f"https://api.openweathermap.org/data/2.5/weather?q={city}&appid=703f64cd37a6b0478592312327805945&units=metric"
  responce = requests.get(url)
  data = responce.json()
  if data["cod"]==200 :
    weather_data = data["main"]
    temperature = weather_data["temp"]
    feelslike = weather_data["feels_like"]
    humidity = weather_data["humidity"]
    pressure = weather_data["pressure"]
    visibility = data["visibility"]
    now = data["dt"]
    sunrise = data["sys"]["sunrise"]
    sunset = data["sys"]["sunset"]
    if now >= sunrise and now <= sunset:
      print("It is currently **daytime** in",city)
    else:
      print("It is currently **night time** in",city)

    return temperature,feelslike,humidity,pressure,visibility
  else:
      raise Exception("Faild to fetch user data")


def main():
  try:
       temperature,feelslike,humidity,pressure,visibility = weather()
       print(f"temperature:{temperature}°C\nfeelslike:{feelslike}°C\nhumidity:{humidity}%\npressure:{pressure}hPa\nvisibility:{visibility}km")
  except Exception as e:
     print(str(e))

if __name__=="__main__":
   main()
